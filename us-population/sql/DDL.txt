select 
	stname 
	, popestimate2010
	, popestimate2020
	, popestimate2020 - popestimate2010 as popestimatediff
	, (popestimate2020 - popestimate2010) /  (popestimate2010 * 1.00) * 100.00 as percdiff
from
	public.us_counties_pop_est_2010_2020_raw
where 
	county = 0;
